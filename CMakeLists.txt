cmake_minimum_required(VERSION 3.25)
project(PRISM VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
option(PRISM_SANITIZE "Enable memory and undefined-behavior checks" OFF)

add_library(prism_options INTERFACE)
target_compile_features(prism_options INTERFACE cxx_std_23)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(prism_options INTERFACE /W4 /WX /permissive-)
else()
    target_compile_options(prism_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(PRISM_SANITIZE)
    if(MSVC)
        message(FATAL_ERROR "The sanitized preset requires a compatible compiler")
    endif()
    target_compile_options(prism_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(prism_options INTERFACE -fsanitize=address,undefined)
endif()

add_executable(prism_check_comments tools/lint/CheckNoComments.cpp)
target_link_libraries(prism_check_comments PRIVATE prism_options)
add_custom_target(lint
    COMMAND prism_check_comments "${CMAKE_CURRENT_SOURCE_DIR}"
    DEPENDS prism_check_comments
)

if(BUILD_TESTING)
    add_executable(prism_foundation_tests tests/unit/FoundationTests.cpp)
    target_link_libraries(prism_foundation_tests PRIVATE prism_options)
    add_test(NAME foundation COMMAND prism_foundation_tests)
    find_package(Threads REQUIRED)
    add_library(prism_alloc_guard STATIC tests/support/AllocGuard.cpp)
    target_link_libraries(prism_alloc_guard PUBLIC prism_options)
    add_executable(prism_allocation_tests tests/unit/AllocationTests.cpp)
    target_link_libraries(prism_allocation_tests PRIVATE prism_alloc_guard Threads::Threads)
    add_test(NAME allocations COMMAND prism_allocation_tests)
    add_library(prism_pixel_oracle STATIC tests/support/PixelOracle.cpp)
    target_link_libraries(prism_pixel_oracle PUBLIC prism_options)
    add_executable(prism_pixel_oracle_tests tests/unit/PixelOracleTests.cpp)
    target_link_libraries(prism_pixel_oracle_tests PRIVATE prism_pixel_oracle prism_alloc_guard)
    add_test(NAME pixel_oracle COMMAND prism_pixel_oracle_tests)
    add_test(NAME no_comments COMMAND prism_check_comments "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME comment_scanner COMMAND prism_check_comments --self-test)
    if(UNIX)
        add_test(NAME commit_messages COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/tools/lint/CheckCommitMsg.sh" --self-test)
    endif()
endif()
