function(prism_test name source)
    add_executable(prism_${name}_tests ${source})
    target_link_libraries(prism_${name}_tests PRIVATE prism_alloc_guard ${ARGN})
    add_test(NAME ${name} COMMAND prism_${name}_tests)
endfunction()
